def calculadora():
  opcao = input("Escolha (1-4): ")
  n1 = float(input("Primeiro número: "))
  n2 = float(input("Segundo número: "))
  if opcao == '1':
    print(f"Resultado: {n1 + n2}")
  elif opcao == '4' and n2 != 0:
    print(f"Resultado: {n1 / n2}")
  else:
    print("Opção inválida!")
